/** 挂在 Leafer Layer.userData.ai 上，随模板 JSON 一起持久化 */
export type LayerRole = 'model' | GarmentRole | 'price' | 'logo' | 'qr' | 'badge' | 'hotspot' | 'deco';
export type GarmentRole = 'garment.top' | 'garment.bottom' | 'garment.full' | 'garment.outer';

export interface AILayerMeta {
  v: 1;
  role: LayerRole;
  // ── 绑定：role=model 时指向"这个模特本季穿什么"，支持表达式
  bind?: {
    skuId?:   string;                 // 精确绑定（主图/KOL 专用模板）
    skuExpr?: 'row.sku.top' | 'row.sku.bottom' | 'row.sku.full'; // 批量列映射
    colorway?: 'row.color' | string;
    modelId?: string;                 // 合成人台 ID（seed 派生，非真人）→ 锁品牌一致性靠 seed 不靠档期
  };
  lockAI?: boolean;                    // true=永不参与生成（LOGO/法务免责语）
  // ── 生成痕迹：可复现 + 可审计 + 可回滚
  lastJobId?: string;
  aiHistory?: { url:string; jobId:string; seed:number; provider:string; at:number; costCent:number }[];
  // ── 热区：role=hotspot 时由 SCHP parsing 回写，相对图层归一化坐标
  bbox?: { x:number; y:number; w:number; h:number; linkSku?:string };
}

// 唯一写入口，禁止业务代码直接摸 userData
export const setRole = (layer:Leaf, role:LayerRole, patch=null) =>
  layer.userData.ai = { v:1, role, ...(layer.userData.ai??{}), ...patch };

/* 模板自检：导出/投产前跑一遍，红项阻断批量 */
export const auditTemplate = (tree:Leaf[]):AuditReport => {
  const c = countByRole(tree);
  return {
    blocking: [
      c.model!==1        && `role=model 必须且只能有 1 个（当前 ${c.model??0}）`,
      !c['garment.top'] && !c['garment.full'] && `缺 garment 图层，AI 试衣无从取衣`,
      tree.some(l=>l.userData.ai?.lockAI!==true && isFontLayer(l) && !isEmbeddedFont(l))
        && `存在未内嵌字体 → 服务端渲染必变宋体/豆腐块`,
    ].filter(Boolean),
    warns: [ !c.hotspot && `无热区图层，详情页不可点击，少一档报价` ].filter(Boolean),
  };
};

③ useGenerate.ts — SSE 订阅 + 图层热替换（含失败语义）

useGenerate.ts · 只拿 job_id，前端零模型 SDKsrc/ai/useGenerate.ts 
import { useAIStore } from '@/stores/ai';

export const useGenerate = () => {
  const ai = useAIStore();

  /** 唯一发起口：先换券拿 jobId，再挂 SSE。同步返回，不 await 结果 */
  const fire = async (layer:Leaf, opts:GenOpts) => {
    const meta = assertBindable(layer);               // role/绑定不齐 → throw 带补救文案
    const q = await api.estimate({ layer:fingerprint(layer), ...opts });
    if (!await confirmCost(q)) return;                  // §01① 费用确认条

    const { jobId, cacheHit } = await api.submit({       // POST /api/jobs → 201
      idemKey : buildIdemKey(layer, opts),            // §03，前端只算不判重
      personId: meta.bind.modelId, garmentSku: resolveSku(layer),
      category : opts.category, mode: opts.mode, seed: opts.seed ?? 20261118,
      segFree  : opts.segFree, templateVer: TEMPLATE_VER, pipelineVer: PIPELINE_VER,
      cbUrl    : layerRef(layer),                      // 回写定位，服务端不认画布
    });
    ai.upsert({ jobId, layerId:layer.id, state: cacheHit?'HIT':'QUEUED', ...q });
    if (cacheHit) ai.setProgress(jobId, 1);            // 命中直接满条，体感 0 等待
    ai.subscribe(jobId);                             // 幂等：同 jobId 不重复挂 SSE
  };

  /** SSE 事件 → 画布反应。这里每一条都对应一个"运营看得见"的变化 */
  const onEvt = (e:JobEvent) => {
    const layer = leafById(e.layerId);
    switch (e.type) {
      case 'queued':   ai.set(e.jobId,{state:'QUEUED', pos:e.pos}); break;   // 角标「排队 37」
      case 'running':   ai.set(e.jobId,{state:'RUNNING', pct:e.pct});
                       layer?..opacity(0.55); showScanline(layer); break;     // 原图半透+扫描线
      case 'preview':   layer?..setAttr('image.url', e.thumbUrl); break;     // 4s 出低清预览，止血焦虑
      case 'postfx':    ai.set(e.jobId,{state:'POSTFX'}); break;            // 「回贴中」= 保 logo 那步
      case 'review':    ai.set(e.jobId,{state:'REVIEW'});
                       commitLayer(layer, e); toast.info('已出图，待审'); break;
      case 'ready':     ai.set(e.jobId,{state:'READY'}); clearBadge(layer); break;
      case 'retry':     ai.set(e.jobId,{state:'RUNNING', try:e.try});
                       toast.warning(`限流，第 ${e.try} 次退避重试中`); break; // 不惊动用户
      case 'failed':    rollbackLayer(layer); ai.set(e.jobId,{state:e.retryable?'RETRY':'DEAD', err:e.err});
                       redBadge(layer, {retry:retryOne, switchProvider:reroute, manual:toDeadLetter}); break;
      case 'quota':     ai.burn(); modal.open('QuotaDrawer', e); break;      // 402 → 熔断+换引擎
    }
  };
  return { fire, onEvt };
};
三条 UX 硬指标 ① 4 秒必须给 preview 事件——人等 6 秒不焦虑，等 6 秒黑屏会连点五下（然后打出五个计费任务）；
② 生成中原图降透明度而不是盖 loading 遮罩，运营要能继续挪旁边的价格图层，画布不许锁；
③ retry 事件不弹 toast 打断，只把角标转成呼吸黄点。退避是系统的事，不是用户的锅。

03幂等键：唯一值得半夜爬起来写的那行代码

def pick_person(sku, slot="hero") -> PersonAsset:
    """前三级全走缓存 → 同 SKU 重出必定同一张脸，这是命中率 +8~12pt 的来源。"""
    if p := binding.hard_person(sku, slot):      return p        # L0 品牌部硬绑（战役主视觉）
    if p := last_used_person(sku, slot, ver_ok=True): return p     # L1 复现优先 ← 最关键的一级
    cand = pool(body_bucket=sku.fit_bucket, tier="prod")            # L2 身材桶不许串
    cand = [p for p in cand if deltaE(p.skin_lab, sku.dominant_lab) > 12]  # 撞色规避
    return weighted_min_use(cand) or fallback_nearest_bucket(sku)   # L3 防"三个爆款脸用到脸盲"

async def try_transition(conn, job_id:int, ev:Event, *, actor:str,
                          payload:dict|None=None, idem:str|None=None) -> bool:
    """唯一允许改 state 的入口。乐观锁 + 合法边校验 + 副作用同事务。"""
    async with conn.transaction():                     # ① 行锁，串行化同 job 的并发事件
        row = await conn.fetchrow(
            "SELECT state, attempts, provider_req_id, cost_est_cent, tenant_id
               FROM tryon_job WHERE id=$1 FOR UPDATE", job_id)
        if row['state'] in TERMINAL:                     # ② 终态幂等：重复 pass/重复 cancel 静默成功
            return False
        if ev not in FROM_ALLOWED[row['state']]:
            raise IllegalTransition(job_id, row['state'], ev)   # ③ 非法边 = bug，告警不吞

        nxt = GUARD[ev](row, payload)                  # ④ 守卫不过抛 BusinessReject（不告警）
        await conn.execute("UPDATE tryon_job SET state=$1, updated_at=now() WHERE id=$2", nxt, job_id)
        await SIDE_EFFECT[ev](conn, row, payload, actor, idem)  # ⑤ 记账/审 log/热区/push 同事务
        await conn.execute("INSERT INTO job_event(job_id,ev,frm,too,actor,at) VALUES($1,$2,$3,$4,$5,now())",
                           job_id, ev, row['state'], nxt, actor)   # ⑥ 全链路可回放
    await bus.publish(f"job.{job_id}", JobEvent(type=nxt, job_id=job_id, ...))  # ⑦ 事务外推 SSE
    return True

class Caps(IntFlag):           # 能力位掩码：路由靠它，不靠 provider 名字字符串
    FLAT_LAY  = 1<<0   # 吃白底平铺图
    WORN_SHOT = 1<<1   # 吃"穿着照"再换（re-context）
    LOCAL_SWAP= 1<<2   # 只换上衣留原裤 ← 只有 OutfitAnyone 有
    SEG_FREE  = 1<<3   # 免 mask
    PARSING   = 1<<4   # 回商品 bbox（热区刚需）
    INPAINT   = 1<<5   # 局部重绘
    CANCEL    = 1<<6
    COMMERCIAL= 1<<7   # ★ NC 协议的模型这一位就是 0，路由直接排除，法务不用逐个审

class TryOnProvider(Protocol):
    def submit  (self, person, garment, cat, opts) -> str: ...   # → provider_req_id
    def poll    (self, req_id) -> tuple[State, Asset|None]: ...
    def cancel  (self, req_id) -> None: ...
    def estimate(self, opts) -> int: ...                      # 分，用于报价条 + 预算闸门
    def caps    (self) -> Caps: ...
    def health  (self) -> float: ...                      # 滑窗 P95 + 失败率 + 429 密度 → 0~1 权重

def route(req, pool) -> Provider:
    ok = [p for p in pool
          if req.caps & p.caps() == req.caps          # ① 能力位必须全满足
          and Caps.COMMERCIAL in p.caps()              # ② 法务硬闸（CatVTON 死在这）
          and p.health() > 0.6                        # ③ 熔断中的不进
          and quota_left(p) > req.estimate()]          # ④ credit/卡时够
    if not ok: raise NoCapableProvider(req, hint='local_swap 仅 OutfitAnyone 支持，可否改为整身换？')
    # ⑤ 成本×健康 加权抽样：便宜的权重高，但抖的就自动让位
    return weighted_pick(ok, w = [p.health() / max(p.estimate(req),1) ** 0.7 for p in ok])

def postfx(raw:np.ndarray, flatlay:np.ndarray, mask:np.ndarray,
           g:GarmentAsset, feat:GarmentFeat) -> PostFXResult:
    H, W = raw.shape[:2]
    # ③ 对齐：TPS 主形变（领/袖/摆/缝 8~12 点对）+ ORB 残差精配
    M = tps_warp(flatlay, feat.kps_src, project_kps(feat.kps_ref, raw.shape))
    M = orb_refine(M, raw, g.logo_bbox)                      # 有印花时这步能救回 1~2px
    if 'stripe' in g.risk_tags: M = phase_align_stripes(M, raw, flatlay, mask)  # E2 专治

    # ④ 分频：σ=7 在 576×864 上约等于"光影 vs 印花"的分界，别照抄到大图
    ai_l  = gauss(raw, 7);   ai_h  = raw - ai_l
    fl_l  = gauss(M,   7);   fl_h  = M - fl_l
    #   低频取 AI（立体感） · 高频取平铺（logo/纱线） · 高频权重按 mask 衰减
    w     = feather(mask, px=9)[..., None] / 255
    if g.logo_bbox: w = boost_in(w, g.logo_bbox, gain=1.0, radius=6)   # logo 区 100% 用原图
    out = ai_l + (0.92*ai_h*(1-w) + 1.04*fl_h*w)                # 1.04 微提纹理，压住"塑料感"

    # ⑤ 缝边：沿 mask 内缩 3px 加方向性阴影，模拟布料包裹
    out = edge_wrap(out, mask, width=3, strength=0.18, light=est_light_dir(raw))
    out = spill_suppress(out, mask, skin_ref=sample_skin(raw, mask), k=0.35)

    # ⑥ ΔE2000 守卫：不过就就地拉回，别指望审核员用眼睛救色差
    lab, base = rgb2lab(mean_in(out, erode(mask,5))), g.base_color_lab
    dE = deltaE2000(lab, base)
    if   dE > 10: return PostFXResult(out, need_review=True, reason='E3', dE=dE)
    elif dE > 6: out = hue_preserving_pull(out, mask, lab, base, amt=0.75)

    # logo 保真硬校验：SSIM<0.92 直接判 E1，不许过审
    s = ssim(crop(out,g.logo_bbox), crop(M,g.logo_bbox))
    return PostFXResult(out, need_review=(s<0.92), reason='E1' if s<0.92 else None,
                          dE=dE, logo_ssim=s, hotspot=bbox_of(mask))
V1 别上"AI 审核 AI" 用一个 VLM 去判"图和货一不一致"，听起来优雅，实际会误杀爆款——它不认识你家的品牌字标，会把正确印花判成变形。V1 就用上面这套确定性规则（ΔE + SSIM + 频域能量 + 印花面积），可解释、可调参、出事能复盘。等攒够 2 万条 review_log，再拿它训一个只做排序（不是判决）的小模型，把人工从全检降到抽检 10%。
